cmake_minimum_required(VERSION 3.22)
project(embedded_jpeg_tests LANGUAGES CXX)
include(FetchContent)
if(POLICY CMP0135)
  cmake_policy(SET CMP0135 NEW)
endif()
FetchContent_Declare(libraw
  URL "${CMAKE_CURRENT_LIST_DIR}/../../third_party/libraw/LibRaw-0.22.3.tar.gz"
  URL_HASH SHA256=789f3c33466210a1a0907f747a97c715d21bbf4af712f257fcbbd90111b227e4)
FetchContent_MakeAvailable(libraw)
file(GLOB_RECURSE raw_sources CONFIGURE_DEPENDS "${libraw_SOURCE_DIR}/src/*.cpp")
# Alternative no-postprocessing placeholders must not duplicate the real implementations.
list(FILTER raw_sources EXCLUDE REGEX "_ph\\.cpp$")
add_library(raw STATIC ${raw_sources})
target_include_directories(raw PUBLIC "${libraw_SOURCE_DIR}")
target_compile_definitions(raw PUBLIC LIBRAW_NOTHREADS LIBRAW_NODLL)
target_compile_features(raw PUBLIC cxx_std_17)
if(MSVC)
  target_compile_definitions(raw PRIVATE _CRT_SECURE_NO_WARNINGS NOMINMAX)
  target_compile_options(raw PRIVATE /utf-8)
endif()
# Host-only contract tests: no codec dependencies, no Android/product target changes.
add_executable(embedded_jpeg_test embedded_jpeg.cpp embedded_jpeg_test.cpp)
target_link_libraries(embedded_jpeg_test PRIVATE raw)
if(MSVC)
  target_compile_options(embedded_jpeg_test PRIVATE /W4 /WX /utf-8)
else()
  target_compile_options(embedded_jpeg_test PRIVATE -Wall -Wextra -Werror)
endif()
enable_testing()
add_test(NAME embedded_jpeg COMMAND embedded_jpeg_test)
