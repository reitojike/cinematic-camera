cmake_minimum_required(VERSION 3.22)
project(halfsize_geometry_tests LANGUAGES CXX)
# Host-only experiment. Android continues to use its one existing LibRaw target.
if(NOT UNIX OR ANDROID)
  message(FATAL_ERROR "This isolated host test requires a POSIX host")
endif()
include(FetchContent)
if(POLICY CMP0135)
  cmake_policy(SET CMP0135 NEW)
endif()
FetchContent_Declare(libraw
  URL "${CMAKE_CURRENT_LIST_DIR}/../../third_party/libraw/LibRaw-0.22.3.tar.gz"
  URL_HASH SHA256=789f3c33466210a1a0907f747a97c715d21bbf4af712f257fcbbd90111b227e4)
FetchContent_MakeAvailable(libraw)
file(GLOB_RECURSE raw_sources CONFIGURE_DEPENDS "${libraw_SOURCE_DIR}/src/*.cpp")
list(FILTER raw_sources EXCLUDE REGEX "_ph\\.cpp$")
add_library(raw STATIC ${raw_sources})
target_include_directories(raw PUBLIC "${libraw_SOURCE_DIR}")
target_compile_definitions(raw PRIVATE LIBRAW_NOTHREADS USE_ZLIB)
target_compile_features(raw PUBLIC cxx_std_17)
find_package(ZLIB REQUIRED)
target_link_libraries(raw PUBLIC ZLIB::ZLIB)
add_executable(halfsize_geometry_test halfsize_geometry_test.cpp
  ../../app/src/main/cpp/original_raster_sink.cpp)
target_include_directories(halfsize_geometry_test PRIVATE ../../app/src/main/cpp)
target_link_libraries(halfsize_geometry_test PRIVATE raw)
target_compile_options(halfsize_geometry_test PRIVATE -Wall -Wextra -Werror)
enable_testing()
add_test(NAME halfsize_geometry COMMAND halfsize_geometry_test)
